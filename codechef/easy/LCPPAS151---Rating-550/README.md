### Print all the arithematic operations

#include <iostream>
using namespace std;

int main() {
    cout<< 8 + 4 << endl;
    cout<< 8 - 4 << endl;
    cout<< 8 / 4 << endl;
    cout<< 8 * 4 << endl;

}
